using System;
using System.Drawing;
using System.Windows.Forms;

namespace VesiqeApp
{
    public partial class Form1 : Form
    {
        private int timerCounter = 0;
        private bool isDataComplete = false;

        public Form1()
        {
            InitializeComponent();
        }

        private void Form1_Load(object sender, EventArgs e)
        {
            progressBar1.Visible = false;
            progressBar1.Value = 0;
            timer1.Interval = 50; // Timer sürəti
        }

        private void txtSeriyaNomer_TextChanged(object sender, EventArgs e)
        {
            lblSeriyaValue.Text = txtSeriyaNomer.Text;
        }

        private void txtFin_TextChanged(object sender, EventArgs e)
        {
            lblFinValue.Text = txtFin.Text;
        }

        private void txtSoyad_TextChanged(object sender, EventArgs e)
        {
            lblSoyadValue.Text = txtSoyad.Text;
        }

        private void txtAd_TextChanged(object sender, EventArgs e)
        {
            lblAdValue.Text = txtAd.Text;
        }

        private void txtAtaAdi_TextChanged(object sender, EventArgs e)
        {
            lblAtaAdiValue.Text = txtAtaAdi.Text;
        }

        private void txtCins_TextChanged(object sender, EventArgs e)
        {
            lblCinsValue.Text = txtCins.Text;
        }

        private void txtDogumTarixi_TextChanged(object sender, EventArgs e)
        {
            lblDogumTarixiValue.Text = txtDogumTarixi.Text;
        }

        private void txtDogumYeri_TextChanged(object sender, EventArgs e)
        {
            lblDogumYeriValue.Text = txtDogumYeri.Text;
        }

        private void txtAileVeziyyeti_TextChanged(object sender, EventArgs e)
        {
            lblAileValue.Text = txtAileVeziyyeti.Text;
        }

        private void txtVerilmeTarixi_TextChanged(object sender, EventArgs e)
        {
            lblVerilmeValue.Text = txtVerilmeTarixi.Text;
        }

        private void txtEtibarlidir_TextChanged(object sender, EventArgs e)
        {
            lblEtibarlidirValue.Text = txtEtibarlidir.Text;
        }


            private void btnSekilYukle_Click(object sender, EventArgs e)
        {
            using (OpenFileDialog ofd = new OpenFileDialog())
            {
                ofd.Filter = "PNG Faylları|*.png";
                if (ofd.ShowDialog() == DialogResult.OK)
                {
                    pictureBoxFoto.Image = Image.FromFile(ofd.FileName);
                    pictureBoxFoto.SizeMode = PictureBoxSizeMode.Zoom;
                }
            }
        }

        private void btnYaddaSaxla_Click(object sender, EventArgs e)
        {
            if (string.IsNullOrWhiteSpace(txtSeriyaNomer.Text) ||
                string.IsNullOrWhiteSpace(txtFin.Text) ||
                string.IsNullOrWhiteSpace(txtSoyad.Text) ||
                string.IsNullOrWhiteSpace(txtAd.Text) ||
                string.IsNullOrWhiteSpace(txtAtaAdi.Text) ||
                string.IsNullOrWhiteSpace(txtCins.Text) ||
                string.IsNullOrWhiteSpace(txtDogumTarixi.Text) ||
                string.IsNullOrWhiteSpace(txtDogumYeri.Text) ||
                string.IsNullOrWhiteSpace(txtAileVeziyyeti.Text) ||
                string.IsNullOrWhiteSpace(txtVerilmeTarixi.Text) ||
                string.IsNullOrWhiteSpace(txtEtibarlidir.Text))
            {
                isDataComplete = false; // 3. Xanalardan biri və ya hamısı boşdur
            }
            else
            {
                isDataComplete = true; // 2. Boşluq yoxdursa
            }

            progressBar1.Visible = true;
            progressBar1.Value = 0;
            timerCounter = 0;
            timer1.Start();
        }

        private void timer1_Tick(object sender, EventArgs e)
        {
            timerCounter += 5;
            if (timerCounter <= 100)
            {
                progressBar1.Value = timerCounter;
            }
            else
            {
                timer1.Stop();

                if (isDataComplete)
                {
                    MessageBox.Show("Məlumatlar uğurla yadda saxlandı!", "Uğurlu", MessageBoxButtons.OK, MessageBoxIcon.Information);
                }
                else
                {                    
                MessageBox.Show("Xəbərdarlıq: Məlumatlar tam daxil edilməyib!", "Diqqət", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                }

                progressBar1.Visible = false;
            }
        }
    }
}
